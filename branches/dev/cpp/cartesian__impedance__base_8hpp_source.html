57inline Matrix6d cartesianGainBlocks(double translational, double rotational) {
58 Matrix6d gains = Matrix6d::Zero();
59 gains.topLeftCorner<3, 3>() = translational * Eigen::Matrix3d::Identity();
60 gains.bottomRightCorner<3, 3>() = rotational * Eigen::Matrix3d::Identity();
61 return gains;
62}

64inline Matrix6d defaultCartesianImpedanceStiffness() { return cartesianGainBlocks(500.0, 50.0); }

69inline Matrix6d defaultCartesianImpedanceDamping(const Matrix6d &stiffness) {
70 if (stiffness.isDiagonal(0.0)) {
71 // cwiseMax guards a negative diagonal, which operatorSqrt would turn into a NaN gain
72 const Vector6d damping = 2.0 * stiffness.diagonal().cwiseMax(0.0).cwiseSqrt();
73 return damping.asDiagonal();
74 }
75 Eigen::SelfAdjointEigenSolver<Matrix6d> solver(stiffness);
76 return 2.0 * solver.operatorSqrt();
77}
