155inline Matrix6d cartesianGainBlocks(double translational, double rotational) {
156 Matrix6d gains = Matrix6d::Zero();
157 gains.topLeftCorner<3, 3>() = translational * Eigen::Matrix3d::Identity();
158 gains.bottomRightCorner<3, 3>() = rotational * Eigen::Matrix3d::Identity();
159 return gains;
160}
161
162inline Matrix6d defaultCartesianImpedanceStiffness() { return cartesianGainBlocks(500.0, 50.0); }
163
167inline Matrix6d defaultCartesianImpedanceDamping(const Matrix6d &stiffness) {
168 if (stiffness.isDiagonal(0.0)) {
169 // cwiseMax guards a negative diagonal, which operatorSqrt would turn into a NaN gain
170 const Vector6d damping = 2.0 * stiffness.diagonal().cwiseMax(0.0).cwiseSqrt();
171 return damping.asDiagonal();
172 }
173 Eigen::SelfAdjointEigenSolver<Matrix6d> solver(stiffness);
174 return 2.0 * solver.operatorSqrt();
175}
