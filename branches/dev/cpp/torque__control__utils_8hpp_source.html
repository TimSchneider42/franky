80inline void validateIsometry(const Affine &transform, const char *name, double tolerance = 1e-6) {
81 validateFinite(transform.matrix(), name);
82 const Eigen::Matrix3d &rotation = transform.linear();
83 if (!(rotation.transpose() * rotation).isApprox(Eigen::Matrix3d::Identity(), tolerance)) {
84 throw std::invalid_argument(std::string(name) + " must have an orthonormal rotation block");
85 }
86 if (std::abs(rotation.determinant() - 1.0) > tolerance) {
87 throw std::invalid_argument(std::string(name) + " must have a right-handed rotation block (det == +1)");
88 }
89 const Eigen::Vector4d last_row = transform.matrix().row(3);
90 if (!last_row.head<3>().isZero(tolerance) || std::abs(last_row[3] - 1.0) > tolerance) {
91 throw std::invalid_argument(std::string(name) + " must have a homogeneous last row of (0, 0, 0, 1)");
92 }

133 void validate() const {
134 validateNonNegativeFinite(max_delta_tau, "safety.max_delta_tau");
135 validateNonNegativeFinite(joint_limit_activation_distance, "safety.joint_limit_activation_distance");
136 validateNonNegativeFinite(joint_limit_stiffness, "safety.joint_limit_stiffness");
137 validateNonNegativeFinite(joint_limit_damping, "safety.joint_limit_damping");
138 validateNonNegativeFinite(joint_limit_max_torque, "safety.joint_limit_max_torque");
139
140 // Repulsion is applied only when both limit vectors are set, so setting one alone disables it
141 // without any indication. That is far more likely to be a mistake than an intent.
142 if (lower_joint_limits.has_value() != upper_joint_limits.has_value()) {
143 throw std::invalid_argument(
144 "safety.lower_joint_limits and safety.upper_joint_limits must be set together; setting only one silently "
145 "disables joint-limit repulsion");
146 }
147 if (!lower_joint_limits.has_value()) return;
148
149 validateFinite(*lower_joint_limits, "safety.lower_joint_limits");
150 validateFinite(*upper_joint_limits, "safety.upper_joint_limits");
151 for (int i = 0; i < 7; ++i) {
152 const double width = (*upper_joint_limits)[i] - (*lower_joint_limits)[i];
153 if (width <= 0.0) {
154 throw std::invalid_argument(
155 "safety.lower_joint_limits must be strictly below safety.upper_joint_limits for every joint");
156 }
157 // Beyond half the interval the lower and upper activation regions overlap, so a joint sitting
158 // in the middle would be pushed by both walls at once.
159 if (2.0 * joint_limit_activation_distance > width) {
160 throw std::invalid_argument(
161 "safety.joint_limit_activation_distance must not exceed half the width of the narrowest joint interval, "
162 "otherwise the lower and upper activation regions overlap");
163 }
164 }
165 }

208 const Vector7d &tau_d_calculated, const Vector7d &tau_reference, double max_delta_tau) {
209 Vector7d tau_d_saturated;

217inline Vector7d computeJointLimitTorque(
218 const Vector7d &q, const Vector7d &dq, const Vector7d &lower_joint_limits, const Vector7d &upper_joint_limits,
219 double joint_limit_activation_distance, double joint_limit_stiffness, double joint_limit_damping,
220 double joint_limit_max_torque) {
221 Vector7d tau_limit = Vector7d::Zero();
222 const double activation_distance = std::max(joint_limit_activation_distance, 1e-6);
223 // std::exp overflows to infinity for a deep penetration or a very small activation distance. Cap it
224 constexpr double kMaxPenetration = 50.0;
225
226 for (size_t i = 0; i < 7; i++) {
227 const double lower_soft_limit = lower_joint_limits[i] + activation_distance;
228 const double upper_soft_limit = upper_joint_limits[i] - activation_distance;
229
230 if (q[i] < lower_soft_limit) {
231 const double penetration = std::min((lower_soft_limit - q[i]) / activation_distance, kMaxPenetration);
232 double tau = joint_limit_stiffness * (std::exp(penetration) - 1.0);
233 if (dq[i] < 0.0) tau += joint_limit_damping * (-dq[i]);
234 tau_limit[i] = std::min(tau, joint_limit_max_torque);
235 } else if (q[i] > upper_soft_limit) {
236 const double penetration = std::min((q[i] - upper_soft_limit) / activation_distance, kMaxPenetration);
237 double tau = -joint_limit_stiffness * (std::exp(penetration) - 1.0);
238 if (dq[i] > 0.0) tau -= joint_limit_damping * dq[i];
239 tau_limit[i] = std::max(tau, -joint_limit_max_torque);
240 }
241 }
242
243 return tau_limit;
244}

246inline Vector7d computeFrictionCompensation(const Vector7d &dq, const FrictionCompensationParams &params) {
247 Vector7d tau = Vector7d::Zero();
248 if ((params.coulomb.array() == 0.0).all() && (params.viscous.array() == 0.0).all()) return tau;
249 const double epsilon = std::max(params.velocity_epsilon, 1e-9);
250 for (int i = 0; i < 7; ++i) {
251 const double limit = std::max(params.max_torque[i], 0.0);
252 const double uncompensated = params.coulomb[i] * std::tanh(dq[i] / epsilon) + params.viscous[i] * dq[i];
253 tau[i] = std::clamp(uncompensated, -limit, limit);
254 }
255 return tau;
256}
